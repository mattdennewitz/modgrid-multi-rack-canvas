# Row Drag and Drop Implementation Plan

> **Superseded in part:** cross-case moves became row swaps that never resize a case (`reorderRow` + `swapRows` replaced `moveRow`). See `docs/superpowers/specs/2026-11-02-row-drag-design.md` for the final behavior.

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Drag a row (with its modules and 3U/1U type) to a new position in its case, or into another case of the same width.

**Architecture:** A pure `moveRow` store action does all the work and returns the same state object when a move is refused, so the UI can colour the drop indicator by comparing `reducer(state, action) !== state`. Two small geometry helpers map the cursor to an insertion point and draw the indicator. `App` owns a separate `rowDrag` state beside the module drag; `CaseView` renders a grip per row and the insertion line.

**Tech Stack:** TypeScript, React 19, Vite, Vitest. `npm test`, `npm run typecheck`.

## Global Constraints

- Spec: `docs/superpowers/specs/2026-11-02-row-drag-design.md`.
- `toIndex` is the insertion point (1 to rows+1 of the target) counted before the row is removed.
- Refuse (return the same state): unequal `hp`; source in a different case with one row; any locked module in the moved row or in any row whose number would change; out-of-range row/index. Same-case `toIndex` of `fromRow` or `fromRow + 1` is a no-op (same state).
- `rowTypes` stays normalised (padded to `rows`, absent when all 3U).
- `npm test` and `npm run typecheck` (covers `src`, `test`; `noUnusedLocals` on) pass at the end of every task.

```ts
/** Insertion point (1 to rows+1) for a row dropped at world-space `wy`: before the first row whose midpoint is below it. */
export function rowInsertIndex(c: RowLayout & Pick<CaseItem, 'y'>, wy: number) {
  const localY = wy - c.y;
  for (let r = 1; r <= c.rows; r++) {
    if (localY < rowTop(c, r) + rowHeight(rowTypeOf(c, r)) / 2) return r;
  }
  return c.rows + 1;
}

/** Y of the drop indicator at insertion point `index`, relative to the case origin, centred in the gap between rows. */
export function insertionLineY(c: RowLayout, index: number) {
  if (index <= c.rows) return rowTop(c, index) - ROW_GAP / 2;
  return rowTop(c, c.rows) + rowHeight(rowTypeOf(c, c.rows)) + ROW_GAP / 2;
}
```

```ts
/**
 * Moves a row, with its modules and 3U/1U type, within a case or into another case of the same width.
 * Returns the same state when the move is refused or changes nothing: see the row-drag spec.
 */
function moveRow(state: AppState, a: { fromCaseId: string; fromRow: number; toCaseId: string; toIndex: number }) {
  const from = state.cases.find((c) => c.id === a.fromCaseId);
  const to = state.cases.find((c) => c.id === a.toCaseId);
  if (!from || !to || from.hp !== to.hp) return state;
  if (a.fromRow < 1 || a.fromRow > from.rows || a.toIndex < 1 || a.toIndex > to.rows + 1) return state;
  const same = from.id === to.id;
  if (!same && from.rows === 1) return state;

  // Where the row lands: in one case, moving down shifts the insertion point up by the removed row.
  const target = same && a.toIndex > a.fromRow ? a.toIndex - 1 : a.toIndex;
  if (same && target === a.fromRow) return state;

  /** Where a placed module goes, or its own placement when its row doesn't move. */
  const renumber = (p: Placement): Placement => {
    if (p.caseId === from.id && p.row === a.fromRow) return { ...p, caseId: to.id, row: target };
    if (same) {
      if (p.caseId !== from.id) return p;
      if (a.fromRow < target && p.row > a.fromRow && p.row <= target) return { ...p, row: p.row - 1 };
      if (target < a.fromRow && p.row >= target && p.row < a.fromRow) return { ...p, row: p.row + 1 };
      return p;
    }
    if (p.caseId === from.id && p.row > a.fromRow) return { ...p, row: p.row - 1 };
    if (p.caseId === to.id && p.row >= target) return { ...p, row: p.row + 1 };
    return p;
  };

  let blocked = false;
  const modules = state.modules.map((m) => {
    if (!m.placement) return m;
    const placement = renumber(m.placement);
    if (placement === m.placement) return m;
    if (m.locked) blocked = true;
    return { ...m, placement };
  });
  if (blocked) return state;

  const types = (c: CaseItem) => Array.from({ length: c.rows }, (_, i) => rowTypeOf(c, i + 1));
  const moved = types(from)[a.fromRow - 1];
  const without = types(from).filter((_, i) => i !== a.fromRow - 1);
  const withMoved = (list: RowType[]) => [...list.slice(0, target - 1), moved, ...list.slice(target - 1)];
  const update = (c: CaseItem, rowTypes: RowType[]): CaseItem => ({
    ...c,
    rows: rowTypes.length,
    rowTypes: normalizeRowTypes(rowTypes, rowTypes.length),
  });

  const cases = state.cases.map((c) => {
    if (same && c.id === from.id) return update(c, withMoved(without));
    if (c.id === from.id) return update(c, without);
    if (c.id === to.id) return update(c, withMoved(types(c)));
    return c;
  });
  return { cases, modules };
}
```

```ts
  /** The case under the cursor, as a row-drop target; null unless it has the dragged row's width. */
  const findRowTarget = (clientX: number, clientY: number, from: { caseId: string; row: number }): RowDrag['target'] => {
    const { state, view } = latest.current;
    const canvasRect = canvasRef.current?.getBoundingClientRect();
    const source = state.cases.find((c) => c.id === from.caseId);
    if (!canvasRect || !source || !inside(canvasRect, clientX, clientY)) return null;
    const wx = (clientX - canvasRect.left - view.x) / view.zoom;
    const wy = (clientY - canvasRect.top - view.y) / view.zoom;
    // Later cases render on top; the first one under the cursor is the only candidate.
    const c = [...state.cases].reverse().find((x) => {
      const { w, h } = caseSize(x);
      return wx >= x.x && wx <= x.x + w && wy >= x.y && wy <= x.y + h;
    });
    if (!c || c.hp !== source.hp) return null;
    const index = rowInsertIndex(c, wy);
    const move: Action = { type: 'moveRow', fromCaseId: from.caseId, fromRow: from.row, toCaseId: c.id, toIndex: index };
    return { caseId: c.id, index, valid: reducer(state, move) !== state };
  };

  const onRowGripPointerDown = (e: React.PointerEvent<HTMLDivElement>, caseId: string, row: number) => {
    if (e.button !== 0) return;
    e.stopPropagation();
    e.preventDefault();
    setRowDrag({ caseId, row, target: findRowTarget(e.clientX, e.clientY, { caseId, row }) });
  };
```
